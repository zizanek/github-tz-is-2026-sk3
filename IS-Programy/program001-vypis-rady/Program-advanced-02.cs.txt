string again = "a";

// Hlavní řídicí cyklus aplikace
while (again.ToLower() == "a")
{
    Console.Clear();
    Console.WriteLine("****************************************");
    Console.WriteLine("***** Vizuální řada čísel (Schody) *****");
    Console.WriteLine("****************************************");
    Console.WriteLine("************* Tomáš Žižka **************");
    Console.WriteLine("****************************************");
    Console.WriteLine();

    // 1. Vstup počáteční hodnoty (s kontrolou na nezáporné číslo kvůli vykreslování)
    Console.Write("Zadejte první číslo řady (kladné celé číslo): ");
    int first;
    while (!int.TryParse(Console.ReadLine(), out first) || first < 0)
    {
        Console.Write("Hodnota musí být celé nezáporné číslo. Zadejte znovu: ");
    }

    // 2. Vstup koncové hodnoty
    Console.Write("Zadejte poslední číslo řady (celé číslo): ");
    int last;
    while (!int.TryParse(Console.ReadLine(), out last))
    {
        Console.Write("Nezadali jste celé číslo. Zadejte poslední číslo znovu: ");
    }

    // 3. Vstup diference s kontrolou na kladnou hodnotu
    Console.Write("Zadejte diferenci / výšku schodu (kladné celé číslo): ");
    int step;
    while (!int.TryParse(Console.ReadLine(), out step) || step <= 0)
    {
        Console.Write("Diference musí být kladné celé číslo větší než 0. Zadejte znovu: ");
    }

    // 4. Rekapitulace zadaných hodnot
    Console.WriteLine();
    Console.WriteLine("========================================");
    Console.WriteLine($"První číslo: {first}; Poslední číslo: {last}; Diference: {step}");
    Console.WriteLine("========================================");
    Console.WriteLine();

    // 5. Výpis řady a vykreslení pestrého schodiště
    Console.WriteLine("Vizuální schodiště (výpis řady):");
    if (first <= last)
    {
        int current = first;
        int rowNumber = 0; // Pomocné číslo řádku pro střídání základních barev
        ConsoleColor originalColor = Console.ForegroundColor;

        while (current <= last)
        {
            // A: Nejprve vypíšeme popisek a svislou čáru v původní barvě
            Console.ForegroundColor = originalColor;
            Console.Write($"{current, 4} | ");

            // B: Vybereme základní barvu pro tento konkrétní řádek (střídáme 3 různé odstíny)
            ConsoleColor rowColor;
            ConsoleColor altColor;

            if (rowNumber % 3 == 0)
            {
                rowColor = ConsoleColor.Blue;
                altColor = ConsoleColor.Cyan;
            }
            else if (rowNumber % 3 == 1)
            {
                rowColor = ConsoleColor.DarkYellow;
                altColor = ConsoleColor.Yellow;
            }
            else
            {
                rowColor = ConsoleColor.DarkGreen;
                altColor = ConsoleColor.Green;
            }

            // C: Vykreslíme bloky vedle sebe se střídáním barev/vzorů na jednom řádku
            for (int i = 0; i < current; i++)
            {
                // Střídáme barvu pro každý jednotlivý block na řádku
                if (i % 2 == 0)
                {
                    Console.ForegroundColor = rowColor;
                }
                else
                {
                    Console.ForegroundColor = altColor;
                }

                // Vykreslíme jeden blok. Můžeme použít '█' nebo střídat '█' a '▓' pro lepší vzor
                Console.Write("█");
            }

            // Odřádkujeme po dokončení celého schodu
            Console.WriteLine();

            current += step;
            rowNumber++;
        }

        // Vrácení barvy konzole na původní hodnotu
        Console.ForegroundColor = originalColor;
    }
    else
    {
        Console.WriteLine("První číslo je větší než poslední. Tento program umí vykreslit pouze vzestupné schodiště.");
    }

    // 6. Dotaz na opakování programu
    Console.WriteLine();
    Console.Write("Pro opakování programu stiskněte 'a' a potvrďte klávesou Enter: ");
    again = Console.ReadLine() ?? "";
}
